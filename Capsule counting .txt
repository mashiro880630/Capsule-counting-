<html lang="zh-TW">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>膠囊進貨驗收計算器</title>
    <!-- 引入 Tailwind CSS -->
    <script src="https://cdn.tailwindcss.com"></script>
    <!-- 引入 React 與 ReactDOM -->
    <script crossorigin src="https://unpkg.com/react@18/umd/react.production.min.js"></script>
    <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>
    <!-- 引入 Babel 用於瀏覽器內編譯 JSX -->
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
    <style>
        body { font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif; }
    </style>
</head>
<body class="bg-gray-50">
    <div id="root"></div>

    <script type="text/babel">
        const { useState, useEffect } = React;

        // --- SVG 圖示組件 ---
        const IconCalculator = ({ className }) => (
            <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className} width="24" height="24"><rect x="4" y="2" width="16" height="20" rx="2" ry="2"></rect><line x1="8" y1="6" x2="16" y2="6"></line><line x1="16" y1="14" x2="16.01" y2="14"></line><line x1="16" y1="10" x2="16.01" y2="10"></line><line x1="16" y1="18" x2="16.01" y2="18"></line><line x1="8" y1="14" x2="8.01" y2="14"></line><line x1="8" y1="10" x2="8.01" y2="10"></line><line x1="12" y1="14" x2="12.01" y2="14"></line><line x1="12" y1="10" x2="12.01" y2="10"></line><line x1="8" y1="18" x2="8.01" y2="18"></line><line x1="12" y1="18" x2="12.01" y2="18"></line></svg>
        );
        const IconInfo = ({ className }) => (
            <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className} width="16" height="16"><circle cx="12" cy="12" r="10"></circle><line x1="12" y1="16" x2="12" y2="12"></line><line x1="12" y1="8" x2="12.01" y2="8"></line></svg>
        );
        const IconCheck = ({ className }) => (
            <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className} width="20" height="20"><path d="M22 11.08V12a10 10 0 1 1-5.93-9.14"></path><polyline points="22 4 12 14.01 9 11.01"></polyline></svg>
        );
        const IconPackage = ({ className }) => (
            <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className} width="20" height="20"><line x1="16.5" y1="9.4" x2="7.5" y2="4.21"></line><path d="M21 16V8a2 2 0 0 0-1-1.73l-7-4a2 2 0 0 0-2 0l-7 4A2 2 0 0 0 3 8v8a2 2 0 0 0 1 1.73l7 4a2 2 0 0 0 2 0l7-4A2 2 0 0 0 21 16z"></path><polyline points="3.27 6.96 12 12.01 20.73 6.96"></polyline><line x1="12" y1="22.08" x2="12" y2="12"></line></svg>
        );
        const IconBox = ({ className }) => (
            <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className} width="20" height="20"><path d="M21 8a2 2 0 0 0-1-1.73l-7-4a2 2 0 0 0-2 0l-7 4A2 2 0 0 0 3 8v8a2 2 0 0 0 1 1.73l7 4a2 2 0 0 0 2 0l7-4A2 2 0 0 0 21 16Z"></path><path d="M3.27 6.96 12 12.01 20.73 6.96"></path><path d="M12 22.08V12"></path></svg>
        );
        const IconScale = ({ className }) => (
            <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className} width="20" height="20"><path d="m16 16 3-8 3 8c-.87.65-1.92 1-3 1s-2.13-.35-3-1Z"></path><path d="m2 16 3-8 3 8c-.87.65-1.92 1-3 1s-2.13-.35-3-1Z"></path><path d="M7 21h10"></path><path d="M12 3v18"></path><path d="M3 7h2c2 0 5-1 7-2 2 1 5 2 7 2h2"></path></svg>
        );
        const IconAlert = ({ className }) => (
            <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className} width="24" height="24"><circle cx="12" cy="12" r="10"></circle><line x1="12" y1="8" x2="12" y2="12"></line><line x1="12" y1="16" x2="12.01" y2="16"></line></svg>
        );

        function App() {
            // --- 基礎設定數據 ---
            const WEIGHT_LARGE_BOX = 0.83; // 大箱子
            const WEIGHT_SMALL_BOX = 0.32; // 小箱子
            const WEIGHT_FOIL_BAG = 0.08;  // 鋁箔袋

            // 一大箱包含兩小箱，兩小箱包含兩鋁箔袋
            const TARE_PER_LARGE_BOX = WEIGHT_LARGE_BOX + (WEIGHT_SMALL_BOX * 2) + (WEIGHT_FOIL_BAG * 2); // 1.63 kg

            // --- 狀態管理 ---
            const [totalCapsules, setTotalCapsules] = useState('');
            const [boxCount, setBoxCount] = useState(1);
            const [boxWeights, setBoxWeights] = useState(['']);
            
            const [hasLoose, setHasLoose] = useState(false);
            const [looseWeight, setLooseWeight] = useState('');
            const [looseMethod, setLooseMethod] = useState('none');
            const [customLooseTare, setCustomLooseTare] = useState('');

            // --- 處理函數 ---
            const handleBoxCountChange = (e) => {
                const val = parseInt(e.target.value);
                const count = isNaN(val) || val < 1 ? 1 : val;
                setBoxCount(count);
                
                setBoxWeights(prev => {
                const newWeights = [...prev];
                if (count > prev.length) {
                    for (let i = prev.length; i < count; i++) newWeights.push('');
                } else if (count < prev.length) {
                    newWeights.splice(count);
                }
                return newWeights;
                });
            };

            const handleBoxWeightChange = (index, value) => {
                const newWeights = [...boxWeights];
                newWeights[index] = value;
                setBoxWeights(newWeights);
            };

            // --- 計算邏輯 ---
            const validBoxWeights = boxWeights.map(w => parseFloat(w) || 0);
            const boxesGrossWeight = validBoxWeights.reduce((a, b) => a + b, 0);
            
            // 計算大箱總扣重 (只要有輸入重量的箱子就扣除包材重)
            const filledBoxesCount = validBoxWeights.filter(w => w > 0).length;
            const boxesTareWeight = filledBoxesCount * TARE_PER_LARGE_BOX;

            let looseTareWeight = 0;
            if (hasLoose) {
                if (looseMethod === 'foil') looseTareWeight = WEIGHT_FOIL_BAG;
                else if (looseMethod === 'smallBoxFoil') looseTareWeight = WEIGHT_SMALL_BOX + WEIGHT_FOIL_BAG;
                else if (looseMethod === 'custom') looseTareWeight = parseFloat(customLooseTare) || 0;
            }

            const looseGrossWeight = hasLoose ? (parseFloat(looseWeight) || 0) : 0;
            const totalGrossWeight = boxesGrossWeight + looseGrossWeight;
            const totalTareWeight = boxesTareWeight + looseTareWeight;
            const totalNetWeight = Math.max(0, totalGrossWeight - totalTareWeight); // 總淨重 (kg)

            const capsulesCount = parseInt(totalCapsules) || 0;
            // 淨重轉換為毫克 (1 kg = 1,000,000 mg)，再除以總粒數
            const weightPerCapsule = capsulesCount > 0 ? ((totalNetWeight * 1000000) / capsulesCount) : 0;

            return (
                <div className="min-h-screen p-4 md:p-8">
                <div className="max-w-6xl mx-auto space-y-6">
                    
                    {/* Header */}
                    <header className="bg-white rounded-2xl shadow-sm p-6 flex items-center justify-between border-l-4 border-blue-600">
                    <div>
                        <h1 className="text-2xl font-bold text-gray-800 flex items-center gap-2">
                        <IconCalculator className="text-blue-600" />
                        膠囊進貨驗收計算器
                        </h1>
                        <p className="text-gray-500 mt-1 text-sm">自動扣除標準包材重量，精準計算每粒膠囊毫克數</p>
                    </div>
                    <div className="hidden md:flex items-center gap-2 text-sm text-blue-700 bg-blue-50 px-4 py-2 rounded-lg">
                        <IconInfo className="text-blue-700" />
                        一大箱標準扣重: {TARE_PER_LARGE_BOX.toFixed(2)} kg
                    </div>
                    </header>

                    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
                    
                    {/* 左側：輸入表單 */}
                    <div className="lg:col-span-2 space-y-6">
                        
                        {/* Section 1: 基礎資料 */}
                        <section className="bg-white rounded-2xl shadow-sm p-6 border border-gray-100">
                        <h2 className="text-lg font-semibold text-gray-800 mb-4 flex items-center gap-2">
                            <IconCheck className="text-blue-500"/>
                            1. 驗收基礎資料
                        </h2>
                        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                            <div>
                            <label className="block text-sm font-medium text-gray-700 mb-1">進貨總粒數 (必填)</label>
                            <div className="relative">
                                <input 
                                type="number" 
                                className="w-full pl-3 pr-12 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none transition-all"
                                placeholder="請輸入這批貨的總粒數"
                                value={totalCapsules}
                                onChange={(e) => setTotalCapsules(e.target.value)}
                                />
                                <span className="absolute right-3 top-2 text-gray-400">粒</span>
                            </div>
                            </div>
                            <div>
                            <label className="block text-sm font-medium text-gray-700 mb-1">進貨大箱數</label>
                            <div className="relative">
                                <input 
                                type="number" 
                                min="1"
                                className="w-full pl-3 pr-12 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none transition-all"
                                value={boxCount}
                                onChange={handleBoxCountChange}
                                />
                                <span className="absolute right-3 top-2 text-gray-400">箱</span>
                            </div>
                            </div>
                        </div>
                        </section>

                        {/* Section 2: 大箱重量登錄 */}
                        <section className="bg-white rounded-2xl shadow-sm p-6 border border-gray-100">
                        <h2 className="text-lg font-semibold text-gray-800 mb-4 flex items-center gap-2">
                            <IconPackage className="text-blue-500"/>
                            2. 每箱實際重量登錄
                        </h2>
                        <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-4 gap-4">
                            {boxWeights.map((weight, index) => (
                            <div key={index} className="bg-gray-50 p-3 rounded-lg border border-gray-200">
                                <label className="block text-xs font-semibold text-gray-500 mb-1">第 {index + 1} 箱</label>
                                <div className="relative">
                                <input 
                                    type="number" 
                                    className="w-full px-2 py-1.5 text-sm border border-gray-300 rounded focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none"
                                    placeholder="重量"
                                    value={weight}
                                    onChange={(e) => handleBoxWeightChange(index, e.target.value)}
                                />
                                <span className="absolute right-2 top-1.5 text-xs text-gray-400">kg</span>
                                </div>
                            </div>
                            ))}
                        </div>
                        </section>

                        {/* Section 3: 散貨資料 */}
                        <section className="bg-white rounded-2xl shadow-sm p-6 border border-gray-100">
                        <div className="flex items-center justify-between mb-4">
                            <h2 className="text-lg font-semibold text-gray-800 flex items-center gap-2">
                            <IconBox className="text-blue-500"/>
                            3. 散貨資料
                            </h2>
                            <label className="flex items-center cursor-pointer">
                            <div className="relative">
                                <input 
                                type="checkbox" 
                                className="sr-only" 
                                checked={hasLoose}
                                onChange={(e) => setHasLoose(e.target.checked)}
                                />
                                <div className={`block w-10 h-6 rounded-full transition-colors ${hasLoose ? 'bg-blue-500' : 'bg-gray-300'}`}></div>
                                <div className={`absolute left-1 top-1 bg-white w-4 h-4 rounded-full transition-transform ${hasLoose ? 'transform translate-x-4' : ''}`}></div>
                            </div>
                            <span className="ml-3 text-sm font-medium text-gray-700">有散貨</span>
                            </label>
                        </div>

                        {hasLoose && (
                            <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mt-4 bg-blue-50/50 p-4 rounded-xl border border-blue-100">
                            <div>
                                <label className="block text-sm font-medium text-gray-700 mb-1">散貨實際總重量</label>
                                <div className="relative">
                                <input 
                                    type="number" 
                                    className="w-full pl-3 pr-12 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none"
                                    placeholder="請輸入重量"
                                    value={looseWeight}
                                    onChange={(e) => setLooseWeight(e.target.value)}
                                />
                                <span className="absolute right-3 top-2 text-gray-400">kg</span>
                                </div>
                            </div>
                            <div>
                                <label className="block text-sm font-medium text-gray-700 mb-1">散貨包裝方式 (自動扣除包材)</label>
                                <select 
                                className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none bg-white"
                                value={looseMethod}
                                onChange={(e) => setLooseMethod(e.target.value)}
                                >
                                <option value="none">無包材 / 純膠囊 (扣重: 0 kg)</option>
                                <option value="foil">僅裝於鋁箔袋 (扣重: 0.08 kg)</option>
                                <option value="smallBoxFoil">小箱子 + 鋁箔袋 (扣重: 0.40 kg)</option>
                                <option value="custom">自訂包材重量...</option>
                                </select>
                            </div>
                            
                            {looseMethod === 'custom' && (
                                <div className="md:col-span-2 mt-2">
                                <label className="block text-sm font-medium text-gray-700 mb-1">自訂散貨包材重量</label>
                                <div className="relative w-full md:w-1/2">
                                    <input 
                                    type="number" 
                                    className="w-full pl-3 pr-12 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none"
                                    placeholder="輸入自訂扣重"
                                    value={customLooseTare}
                                    onChange={(e) => setCustomLooseTare(e.target.value)}
                                    />
                                    <span className="absolute right-3 top-2 text-gray-400">kg</span>
                                </div>
                                </div>
                            )}
                            </div>
                        )}
                        </section>
                    </div>

                    {/* 右側：計算結果面板 */}
                    <div className="space-y-6">
                        <div className="bg-gray-800 rounded-2xl shadow-lg p-6 text-white sticky top-6">
                        <h2 className="text-lg font-semibold mb-6 flex items-center gap-2 border-b border-gray-700 pb-3">
                            <IconScale className="text-blue-400"/>
                            結算數據
                        </h2>
                        
                        <div className="space-y-4 text-sm">
                            <div className="flex justify-between items-center text-gray-300">
                            <span>貨物總毛重</span>
                            <span className="font-mono text-lg">{totalGrossWeight.toFixed(2)} <span className="text-xs">kg</span></span>
                            </div>
                            
                            <div className="flex justify-between items-center text-gray-400 border-l-2 border-gray-600 pl-2">
                            <span>- 大箱包材扣重 ({filledBoxesCount}箱)</span>
                            <span className="font-mono">{boxesTareWeight.toFixed(2)} <span className="text-xs">kg</span></span>
                            </div>

                            {hasLoose && (
                            <div className="flex justify-between items-center text-gray-400 border-l-2 border-gray-600 pl-2">
                                <span>- 散貨包材扣重</span>
                                <span className="font-mono">{looseTareWeight.toFixed(2)} <span className="text-xs">kg</span></span>
                            </div>
                            )}
                            
                            <div className="pt-4 border-t border-gray-700 flex justify-between items-center">
                            <span className="font-medium text-blue-200">實際膠囊總淨重</span>
                            <span className="font-mono text-xl font-semibold text-blue-400">{totalNetWeight.toFixed(2)} <span className="text-sm">kg</span></span>
                            </div>
                        </div>

                        <div className="mt-8 bg-gray-900 rounded-xl p-5 border border-gray-700 text-center relative overflow-hidden">
                            <div className="absolute top-0 left-0 w-full h-1 bg-blue-500"></div>
                            <h3 className="text-gray-400 text-sm font-medium mb-2">平均每粒膠囊重量</h3>
                            {capsulesCount > 0 ? (
                            <div className="flex items-end justify-center gap-1">
                                <span className="text-4xl md:text-5xl font-bold text-white tracking-tight">
                                {weightPerCapsule.toFixed(1)}
                                </span>
                                <span className="text-blue-400 font-medium mb-1">mg</span>
                            </div>
                            ) : (
                            <div className="flex flex-col items-center justify-center py-4 text-yellow-500 gap-2">
                                <IconAlert className="text-yellow-500" />
                                <span className="text-sm">請輸入進貨總粒數</span>
                            </div>
                            )}
                        </div>
                        
                        {/* 包材說明 */}
                        <div className="mt-6 bg-gray-700/50 rounded-lg p-4 text-xs text-gray-400 space-y-1">
                            <p className="font-semibold text-gray-300 mb-2">預設包材基準數據：</p>
                            <div className="flex justify-between"><span>大箱子</span><span>0.83 kg / 個</span></div>
                            <div className="flex justify-between"><span>小箱子</span><span>0.32 kg / 個</span></div>
                            <div className="flex justify-between"><span>鋁箔袋</span><span>0.08 kg / 個</span></div>
                            <div className="border-t border-gray-600 my-1 pt-1 flex justify-between text-gray-300">
                            <span>單一大箱總扣重</span>
                            <span>1.63 kg / 箱</span>
                            </div>
                        </div>

                        </div>
                    </div>

                    </div>
                </div>
                </div>
            );
        }

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
</body>
</html>
